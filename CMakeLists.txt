cmake_minimum_required(VERSION 3.24)
project(kernel-manager VERSION 0.1.0 LANGUAGES C CXX)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE MinSizeRel CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(KERNEL_SOURCE "/usr/src/linux" CACHE PATH "Linux source tree providing the native Kconfig engine")
set(KCONFIG_DIR "${KERNEL_SOURCE}/scripts/kconfig")
if(NOT EXISTS "${KCONFIG_DIR}/parser.y")
  message(FATAL_ERROR "Set KERNEL_SOURCE to a complete Linux source tree")
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Svg)
find_package(KF6WindowSystem QUIET)
find_package(BISON REQUIRED)
find_package(FLEX REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/kconfig")
BISON_TARGET(KconfigParser "${KCONFIG_DIR}/parser.y" "${CMAKE_CURRENT_BINARY_DIR}/kconfig/parser.tab.c"
             DEFINES_FILE "${CMAKE_CURRENT_BINARY_DIR}/kconfig/parser.tab.h")
FLEX_TARGET(KconfigLexer "${KCONFIG_DIR}/lexer.l" "${CMAKE_CURRENT_BINARY_DIR}/kconfig/lexer.lex.c")
ADD_FLEX_BISON_DEPENDENCY(KconfigLexer KconfigParser)
add_library(kconfig STATIC
  "${KCONFIG_DIR}/confdata.c" "${KCONFIG_DIR}/expr.c" "${KCONFIG_DIR}/menu.c"
  "${KCONFIG_DIR}/preprocess.c" "${KCONFIG_DIR}/symbol.c" "${KCONFIG_DIR}/util.c"
  ${BISON_KconfigParser_OUTPUTS} ${FLEX_KconfigLexer_OUTPUTS})
set_target_properties(kconfig PROPERTIES AUTOMOC OFF)
target_compile_definitions(kconfig PRIVATE YYDEBUG=1)
target_include_directories(kconfig PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/kconfig" "${KCONFIG_DIR}" "${KERNEL_SOURCE}/scripts/include")
add_executable(kernel-manager src/main.cpp src/engine.cpp src/catalog.cpp src/window.cpp src/jobs.cpp src/insights.cpp src/theme.cpp src/workbench.cpp src/eventbus.cpp src/session.cpp src/backend.cpp src/menumodel.cpp src/asyncui.cpp src/icons.cpp)
target_link_libraries(kernel-manager PRIVATE kconfig Qt6::Widgets Qt6::Network Qt6::Svg)
if(TARGET KF6::WindowSystem)
  target_link_libraries(kernel-manager PRIVATE KF6::WindowSystem)
  target_compile_definitions(kernel-manager PRIVATE KM_HAVE_WINDOW_EFFECTS)
endif()
file(GLOB KM_ICONS CONFIGURE_DEPENDS "assets/*.svg" "assets/distro/*.svg")
qt_add_resources(kernel-manager "icons" PREFIX "/assets" BASE "assets" FILES ${KM_ICONS})
target_compile_options(kernel-manager PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(kernel-manager PRIVATE
  KM_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" KM_KERNEL_SOURCE="${KERNEL_SOURCE}")
install(TARGETS kernel-manager RUNTIME DESTINATION bin)
install(DIRECTORY assets catalog DESTINATION share/kernel-manager PATTERN cache EXCLUDE)
